{# php/ui 渲染层的共用表格宏。 对应 PHP 的 EchoTableParagraphBegin() / EchoTableColumn() / EchoTableParagraphEnd() 三件套：那边是"开始 echo 表头 → 每行 echo → 结束 echo"，这里是一个宏。 为什么抽出来：php/ui 里还有十几个段落（trading / stock / sma / portfolio …） 都是同一套"列定义 + 行数据"，逐个在模板里复制这段循环会让每加一个段落 都重复二十行 Jinja。列头颜色、单元格颜色、首格外链这三条规则集中在这里改。 table —— service 返回的 {"columns", "rows", "id"} 结构 css_class —— 追加在 data-table 之后的页面专属类 first_cell_is_html —— 首格是否为 HTML（PHP 那里首格恒为 GetStockLink() 之类的 ）。 默认 False：只有确实需要透传标签的表格才打开， 避免把业务文本也一并关掉转义。 #} {% macro data_table(table, css_class="", first_cell_is_html=False) %} {% if table %} {% for column in table.columns %} {% if column.color %}{{ column.text }}{% else %}{{ column.text }}{% endif %} {% endfor %} {% for row in table.rows %} {# row.background 是整行底色（PHP 的 background-color）；row.href 只作用于首格 #} {% for cell in row.cells %} {% set text = cell if cell is string else cell[0] %} {% set color = "" if cell is string else cell[1] %} {% set hinted = loop.first and row.title %} {% set index = loop.index0 %} {% set classes = ("cell-" ~ color if color else "") ~ (" with-hint" if hinted else "") %} {%- if loop.first and first_cell_is_html -%} {{ text|safe }} {%- elif row.hrefs.get(index) is string -%} {# 非首格也能挂单个链接：URL 在这里包一层 #} {{ text }} {%- elif row.hrefs.get(index) -%} {# 一串链接（管理员的"编辑/删除"）：每项的 text 与 url 都来自数据 #} {%- for link in row.hrefs.get(index) -%} {{ link.text }}{% if not loop.last %} {% endif %} {%- endfor -%} {%- elif loop.first and row.href -%} {{ text }} {%- else -%} {{ text }} {%- endif -%} {% endfor %} {% endfor %}
{% endif %} {% endmacro %}
